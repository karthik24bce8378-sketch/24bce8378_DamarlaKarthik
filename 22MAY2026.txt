dname VARCHAR(25) NOT NULL UNIQUE );

SELECT * FROM departments 
CROSS JOIN hod 
order by did asc;

SELECT *FROM departments d 
INNER JOIN hod h
WHERE d.did=h.hid;

SELECT *FROM departments d 
INNER JOIN hod h
ON d.did=h.hid;

SELECT d.dname, h.hname FROM departments d 
INNER JOIN hod h
ON d.did=h.hid;

SELECT *FROM departments d NATURAL JOIN hod h;

SELECT *FROM departments d 
LEFT JOIN hod h ON d.did=h.hid
ORDER BY did ASC;

SELECT d.dname, h.hname FROM departments d 
LEFT JOIN hod h ON d.did=h.hid
ORDER BY dname ASC;

SELECT * FROM departments d 
RIGHT JOIN hod h ON d.did=h.hid
ORDER BY did ASC;
