SELECT FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE DEPARTMENT='Admin' AND SALARY>200000;

SELECT FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE DEPARTMENT='Account' OR SALARY>400000;

SELECT WORKER_ID, FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE WORKER_ID IN (2,5,7);

SELECT WORKER_ID, FIRST_NAME, DEPARTMENT, SALARY
FROM Worker
WHERE WORKER_ID NOT IN (1,3,4);

SELECT *
FROM Worker
WHERE SALARY BETWEEN 50000 AND 300000;

SELECT *
FROM Worker
WHERE SALARY NOT BETWEEN 150000 AND 350000;

SELECT *
FROM Worker
WHERE SALARY BETWEEN 100000 AND 600000
      AND DEPARTMENT NOT IN ('Admin','HR');

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE 'S%';

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE '%N';

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE '_a%';

SELECT *
FROM Worker
WHERE FIRST_NAME LIKE '%an%';

SELECT *
FROM Worker
WHERE DEPARTMENT='Account'
      AND SALARY IN (
            SELECT MIN(SALARY)
            FROM Worker
            WHERE DEPARTMENT='Account'
                  AND FIRST_NAME NOT LIKE 'S%'
      )
      AND FIRST_NAME NOT LIKE 'S%';

SELECT MIN(SALARY)
FROM Worker
WHERE DEPARTMENT='Admin';

SELECT MAX(SALARY)
FROM Worker
WHERE DEPARTMENT='Account';

SELECT SUM(SALARY)
FROM Worker
WHERE DEPARTMENT='Admin';

SELECT COUNT(WORKER_ID)
FROM Worker
WHERE DEPARTMENT='HR';

SELECT AVG(SALARY)
FROM Worker
WHERE DEPARTMENT='Admin';

SELECT WORKER_ID AS EMPLOYEE_NUMBER
FROM Worker;

SELECT DISTINCT(DEPARTMENT)
FROM Worker;

SELECT WORKER_ID, FIRST_NAME, DEPARTMENT,
CASE
      WHEN SALARY > 500000
            THEN 'Very High Salary'
      WHEN SALARY BETWEEN 200000 AND 500000
            THEN 'Moderate Salary'
      ELSE 'Low Salary'
END
AS SALARY_STATUS
FROM Worker;

SELECT *
FROM Worker;

CREATE VIEW MinSalariedWorkers AS
SELECT FIRST_NAME, LAST_NAME, SALARY, DEPARTMENT
FROM Worker
WHERE SALARY IN (
      SELECT MIN(SALARY)
      FROM Worker
      GROUP BY DEPARTMENT
);

CREATE VIEW MaxSalariedWorkers AS
SELECT FIRST_NAME, LAST_NAME, SALARY, DEPARTMENT
FROM Worker
WHERE SALARY IN (
      SELECT MAX(SALARY)
      FROM Worker
      GROUP BY DEPARTMENT
);

SELECT *
FROM MinSalariedWorkers;

SELECT *
FROM MaxSalariedWorkers;
