});


6. Find employees with salary equal to 50000. 

db.employees.find({
    salary: 50000
});


7. Find employees with salary less than 60000. 

db.employees.find({
    salary: {$lt: 60000}
});


8. Find employees with salary greater than or equal to 60000. 

db.employees.find({
    salary: {$gte: 60000}
});


9. Find employees whose age is not equal to 30. 

db.employees.find({
    age: {$ne: 30}
});


10. Find employees from Chennai or Mumbai. 

db.employees.find({
    $or: [
        {city: "Chennai"},
        {city: "Mumbai"}
    ]
});


11. Find employees not from Chennai. 

db.employees.find({
  city: {$ne: "Chennai"}
});


12. Find IT employees with salary > 50000.

db.employees.find({
  department: "IT",
  salary: {$gt: 50000}
});


13. Find employees from Chennai OR Bangalore.

db.employees.find({
    city: {
        $in: ["Chennai", "Bangalore"]
    }
});


14. Find employees NOT earning above 60000.

db.employees.find({
  salary: {
    $not : {$gt: 60000}
  }
});


15. Find employees neither in IT nor HR.

db.employees.find({
    department: {
        $nin: ["IT", "HR"]
    }
});


16. Find documents containing experience field.

db.employees.find({
  experience: {$exists: true}
});


17. Find documents where salary is Integer.

db.employees.find({
  salary: {
    $type: "int"
  }
});


18. Find employees having MongoDB skill.

db.employees.find({
    skills: "MongoDB"
});


19. Find employees having both MongoDB and AWS.

db.employees.find({
    skills: {
        $all: ["MongoDB", "AWS"]
    }
});


20. Find employees with exactly 3 skills.

db.employees.find({
    skills: {
        $size: 3
    }
});


21. Find employees having Java skill using elemMatch.

db.employees.find({
    skills: {
        $elemMatch: {
            $eq: "Java"
        }
    }
});


22. Display employees sorted by salary descending.

db.employees.find().sort({
    salary: -1
});


23. Display top 3 highest paid employees.

db.employees.find().sort({
    salary: -1
}).limit(3);


24. Increase salary of employee 101 by 5000.

db.employees.updateOne(
    { empId: 101 },
    {$inc: {salary: 5000}
});


25. Increase salary of all IT employees by 10%.

db.employees.updateMany(
  {department: "IT"},
  [{
    $set: {
      salary: {
        $multiply: ["$salary", 1.10]
      }
    }
  }]
);
